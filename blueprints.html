/* ========================= CANVAS PAN + PINCH ========================= */ let pointers = new Map(); let isPanning = false; let panStartX = 0; let panStartY = 0; let panStartPanX = 0; let panStartPanY = 0; let pinchStartDistance = 0; let pinchStartScale = 1; let pinchCenterX = 0; let pinchCenterY = 0; /* ========================= POINTER DOWN ========================= */ viewport.addEventListener( "pointerdown", (e) => { /* 写真を触った場合は 写真のドラッグを優先 */ if ( e.target.classList.contains("drag") ) { return; } pointers.set( e.pointerId, { x: e.clientX, y: e.clientY } ); /* ===================== 1本指 PAN ===================== */ if (pointers.size === 1) { isPanning = true; panStartX = e.clientX; panStartY = e.clientY; panStartPanX = panX; panStartPanY = panY; } /* ===================== 2本指 PINCH ===================== */ if (pointers.size === 2) { isPanning = false; const pts = [...pointers.values()]; pinchStartDistance = Math.hypot( pts[0].x - pts[1].x, pts[0].y - pts[1].y ); pinchStartScale = scale; pinchCenterX = (pts[0].x + pts[1].x) / 2; pinchCenterY = (pts[0].y + pts[1].y) / 2; } } ); /* ========================= POINTER MOVE ========================= */ viewport.addEventListener( "pointermove", (e) => { if ( !pointers.has(e.pointerId) ) { return; } pointers.set( e.pointerId, { x: e.clientX, y: e.clientY } ); /* ===================== 1本指 PAN ===================== */ if ( pointers.size === 1 && isPanning ) { panX = panStartPanX + (e.clientX - panStartX); panY = panStartPanY + (e.clientY - panStartY); updateCanvas(); return; } /* ===================== 2本指 PINCH ===================== */ if ( pointers.size === 2 ) { const pts = [...pointers.values()]; const distance = Math.hypot( pts[0].x - pts[1].x, pts[0].y - pts[1].y ); let newScale = pinchStartScale * ( distance / pinchStartDistance ); newScale = Math.max( 0.3, Math.min(4, newScale) ); /* ピンチ中心を基準に 拡大縮小 */ const worldX = (pinchCenterX - panX) / scale; const worldY = (pinchCenterY - panY) / scale; scale = newScale; panX = pinchCenterX - worldX * scale; panY = pinchCenterY - worldY * scale; updateCanvas(); } } ); /* ========================= POINTER UP ========================= */ viewport.addEventListener( "pointerup", (e) => { pointers.delete( e.pointerId ); if ( pointers.size === 0 ) { isPanning = false; } /* 2本指から1本指に戻った時 新しいPANを開始 */ if ( pointers.size === 1 ) { const remaining = [...pointers.values()][0]; isPanning = true; panStartX = remaining.x; panStartY = remaining.y; panStartPanX = panX; panStartPanY = panY; } } ); /* ========================= POINTER CANCEL ========================= */ viewport.addEventListener( "pointercancel", (e) => { pointers.delete( e.pointerId ); if ( pointers.size === 0 ) { isPanning = false; } } );
